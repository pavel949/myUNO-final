model User {
  id                String            @id @default(uuid())
  email             String?           @unique
  phoneNumber       String?           @unique
  firstName         String
  lastName          String
  avatarUrl         String?
  role              UserRole          @default(GUEST)
  identityProvider  IdentityProvider  @default(EMAIL_OTP)
  locale            String            @default("ru")
  currencyPreference String           @default("THB") // THB, USD, EUR, RUB
  isActive          Boolean           @default(true)
  isVerified        Boolean           @default(false)
  passportNumber    String?
  passportCountry   String?
  createdAt         DateTime          @default(now())
  updatedAt         DateTime          @updatedAt

  // Связи
  sessions          Session[]
  auditLogs         AuditLog[]
  ownedUnits        Unit[]            @relation("UnitOwner")
  bookings          Booking[]         @relation("GuestBookings")
  serviceOrders     ServiceOrder[]    @relation("GuestServiceOrders")
  assignedInspections CleaningInspection[] @relation("Inspector")
  assignedCleanings  HousekeepingTask[]   @relation("Housekeeper")
  reportedTickets   MaintenanceTicket[]  @relation("TicketReporter")
  assignedTickets   MaintenanceTicket[]  @relation("TicketAssignee")
  agentDeals        AgentDeal[]       @relation("AgentUser")
  vendorProfile     VendorPartner?
  payoutAccounts    PayoutAccount[]

  @@index([role, isActive])
  @@index([email])
  @@index([phoneNumber])
}

model Session {
  id           String   @id @default(uuid())
  userId       String
  token        String   @unique
  userAgent    String?
  ipAddress    String?
  expiresAt    DateTime
  createdAt    DateTime @default(now())

  user         User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
}

model AuditLog {
  id           String   @id @default(uuid())
  userId       String?
  action       String   // e.g., "UNIT_PRICE_UPDATE", "CHECK_IN_EXPRESS", "SPLIT_PAYOUT_TRIGGERED"
  entityType   String   // "Unit", "Booking", "LedgerTransaction", "OperatingMandate"
  entityId     String
  diffJson     Json?    // Предыдущие и новые значения
  ipAddress    String?
  timestamp    DateTime @default(now())

  user         User?    @relation(fields: [userId], references: [id], onDelete: SetNull)

  @@index([entityType, entityId])
  @@index([timestamp])
}
