async function flow(method, start, end, tag) {
  await gp.goto(`${BASE}/units/${UNIT}?startDate=${start}&endDate=${end}&adults=2&children=0`); await gp.waitForLoadState('networkidle');
  check(`${tag} unit page shows total price`, /Total ฿[\d,]+/.test(await text(gp)));
  await gp.getByRole('button',{name:/request to book|reserve/i}).first().click();
  await gp.waitForURL(/book\/review/); await gp.waitForLoadState('networkidle');
  await gp.locator('input[type=checkbox]').check();
  await gp.locator('select').filter({hasText:'Cash on arrival'}).selectOption({label: method==='cash'?'Cash on arrival':'Bank transfer'});
  await gp.getByRole('button',{name:'Confirm booking'}).click();
  await gp.waitForURL(/\/trips/,{timeout:20000}); await gp.waitForLoadState('networkidle');
  const row=sql(`select id||'|'||status||'|'||payment_method||'|'||(hold_expires_at is null) from booking where start_date='${start}' order by created_at desc limit 1`).split('|');
  const bid=row[0];
  check(`${tag} request created (requested, ${method}, no hold yet)`, row[1]==='requested' && row[2]===(method==='cash'?'cash':'bank_transfer') && row[3]==='t', row.join(','));
  // approve
  await ap.goto(BASE+'/ops/requests'); await ap.waitForLoadState('networkidle');
  await ap.getByRole('button',{name:'Approve'}).first().click(); await ap.waitForTimeout(2500);
  const h=sql(`select status||'|'||round(extract(epoch from (hold_expires_at-now()))/3600) from booking where id='${bid}'`);
  check(`${tag} host approves → pending_payment, ~24h hold`, h==='pending_payment|24', h);
  await gp.goto(`${BASE}/trips/${bid}`); await gp.waitForLoadState('networkidle'); await gp.waitForTimeout(1500);
  const t=await text(gp);
  await gp.screenshot({path:OUT+`12-${tag}-awaiting-payment.png`,fullPage:true});
  check(`${tag} guest sees only the chosen rail`, method==='cash' ? (!/Pay by card/.test(t) && !/Payment reference/.test(t) && /cash to our team/i.test(t)) : (!/Pay by card/.test(t) && /Payment reference/.test(t)), '');
  check(`${tag} no dispute panel before any charge`, !/Dispute this charge/.test(t));
  if (method==='bank_transfer') {
    const m=t.match(/within (\d+)h/); check('B transfer deadline ≤ 24h hold', m && +m[1]<=24, m?.[0]);
  }
  return bid;
}
